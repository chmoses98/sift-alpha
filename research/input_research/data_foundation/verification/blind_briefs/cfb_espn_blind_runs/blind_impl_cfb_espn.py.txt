def toint(s):
    return int(str(s).strip())
def parse(path):
    try:
        j=json.load(open(path))
        co=j['header']['competitions'][0]
        if co['status']['type']['completed'] is not True: return None
        H=A=None
        for c in co['competitors']:
            if c['homeAway']=='home': H=c
            elif c['homeAway']=='away': A=c
        hs=toint(H['score']); as_=toint(A['score'])
        lq={}
        for k,c in (('h',H),('a',A)):
            ls=c['linescores']
            if len(ls)<4: return None
            lq[k]=[toint(x['displayValue']) for x in ls[:4]]
        nper=max(len(H['linescores']),len(A['linescores']))
        gid=toint(j['header']['id']); season=int(j['header']['season']['year'])
        ht=toint(H['team']['id']); at=toint(A['team']['id'])
        d=co['date']
        dt=datetime.datetime.strptime(d,'%Y-%m-%dT%H:%MZ')-datetime.timedelta(hours=10)
        date=dt.date().isoformat()
        ne=0 if co.get('neutralSite') is True else 1
    except Exception:
        return None
    r=dict(gid=gid,season=season,ht=ht,at=at,date=date,ne=ne,hq=lq['h'],aq=lq['a'],hs=hs,as_=as_,nper=nper)
    sp=j.get('scoringPlays') or []
    tds=[]
    for e in sp:
        try:
            if (e.get('scoringType') or {}).get('name')=='touchdown':
                tds.append(toint(e['team']['id']))
        except Exception:
            tds.append(None)
    nan=float('nan')
    if len(sp)>0 or (hs==0 and as_==0):
        r['htd']=sum(1 for x in tds if x==ht); r['atd']=sum(1 for x in tds if x==at)
        if not tds: r['ftd']=0
        else: r['ftd']=1 if tds[0]==ht else (-1 if tds[0]==at else nan)
    else:
        r['htd']=r['atd']=r['ftd']=nan
    # box
    try:
        st={}
        for t in j['boxscore']['teams']:
            tid=toint(t['team']['id'])
            s={x['name']:x['displayValue'] for x in t['statistics']}
            yd=int(s['totalYards']); ru=int(s['rushingAttempts'])
            ca=re.split(r'[/-]',s['completionAttempts'].strip())
            pl=int(ca[1])+ru
            st[tid]=(yd,pl,ru)
        a=st[ht];b=st[at]
        if not(a[1]>0 and b[1]>0): raise ValueError
        r['hy'],r['hp'],r['hr']=a; r['ay'],r['ap'],r['ar']=b
    except Exception:
        r['hy']=r['hp']=r['hr']=r['ay']=r['ap']=r['ar']=nan
    try:
        dr=j['drives']['previous']
        hd=ad=0
        for x in dr:
            try: tid=toint(x['team']['id'])
            except Exception: continue
            if tid==ht: hd+=1
            elif tid==at: ad+=1
        if hd==0 or ad==0: raise ValueError
        r['hd']=hd;r['ad']=ad
    except Exception:
        r['hd']=r['ad']=nan
    return r

def kernel(xs,bws,events,minref):
    """xs list of arrays; events dict name->(v, mask) ; returns dict name->array"""
    N=2*n
    xp=np.vstack(xs).T; okx=~np.isnan(xp).any(1)
    names=list(events); V=np.column_stack([np.where(events[k][1],events[k][0],0.0) for k in names])
    M=np.column_stack([events[k][1] for k in names]).astype(float)
    res=np.full((N,len(names)),nan)
    for s in np.unique(season):
        q=np.nonzero((season==s)&okx)[0]
        ref=np.nonzero((season<s)&okx)[0]
        if len(q)==0 or len(ref)==0: continue
        okev=M[ref].sum(0)>=minref
        if not okev.any(): continue
        Vr=V[ref];Mr=M[ref]
        for c0 in range(0,len(q),1500):
            qq=q[c0:c0+1500]
            S=np.zeros((len(qq),len(ref)))
            for dim,bw in enumerate(bws):
                S+=((xp[ref,dim][None,:]-xp[qq,dim][:,None])/bw)**2
            K=np.exp(-0.5*S)
            num=K@Vr; den=K@Mr
            with np.errstate(all='ignore'):
                val=num/den
            val[:,~okev]=nan; val[den==0]=nan
            res[qq]=val
    return {k:res[:,i] for i,k in enumerate(names)}
