def parse(x):
    if x is None or (isinstance(x, float) and np.isnan(x)):
        return None
    items = re.findall(r'\{[^}]*\}', str(x))
    out = []
    for pos, it in enumerate(items, 1):
        mv = re.search(r"'value':\s*([-+0-9.eE]+|nan)", it)
        if mv is None:
            continue
        v = float(mv.group(1))
        mp = re.search(r"'period':\s*([-+0-9.eE]+)", it)
        p = float(mp.group(1)) if mp else float(pos)
        out.append((p, v))
    out.sort(key=lambda a: a[0])
    return [v for _, v in out]

def Dn(y, m, s):
    return ndtr((y + 0.5 - m) / s) - ndtr((y - 0.5 - m) / s)


def fit(y, mean, ep, pair):
    best = (-np.inf, None)
    r = y - mean
    rms = np.sqrt(np.mean(r ** 2))
    J = np.arange(-10, 11)
    for t1 in (0, 0.5, 1):
        for gg in (0, 0.5, 1):
            base = (ep / 100) ** t1 * pair ** gg
            sref = rms / np.mean(base)
            s0s = sref * 1.05 ** J
            S = s0s[:, None] * base[None, :]
            d1 = Dn(y[None, :], mean[None, :], S)
            for w in (0, 0.1, 0.2, 0.3):
                for k in (1.5, 2, 3):
                    d2 = Dn(y[None, :], mean[None, :], k * S)
                    P = (1 - w) * d1 + w * d2
                    ll = np.log(np.maximum(P, 1e-300)).sum(axis=1)
                    for jj in range(len(J)):
                        if ll[jj] > best[0]:
                            best = (ll[jj], (s0s[jj], t1, gg, w, k))
    return best[1]
