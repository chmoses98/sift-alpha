def E(col, H, m):
    return t.groupby('team', sort=False)[col].transform(lambda s: s.shift(1).ewm(halflife=H, min_periods=m).mean())

def before_date(mask, vals):
    """per-row (sum, count) of vals over rows with mask & date < row date."""
    v = pd.Series(np.where(mask, vals, 0.0)); n = pd.Series(mask.astype(float))
    gs = v.groupby(t.date).sum().sort_index(); gn = n.groupby(t.date).sum().sort_index()
    cs = gs.cumsum().shift(1).fillna(0.0); cn = gn.cumsum().shift(1).fillna(0.0)
    return t.date.map(cs).values, t.date.map(cn).values

def mean_before(mask, vals):
    s, n = before_date(mask, vals)
    return np.where(n > 0, s / np.where(n > 0, n, 1), np.nan), n

def team_prior(mask, vals):
    """per-row (sum, count) of vals over the team's earlier rows with mask."""
    v = pd.Series(np.where(mask, vals, 0.0)); n = pd.Series(mask.astype(float))
    s = v.groupby(t.team).cumsum() - v
    c = n.groupby(t.team).cumsum() - n
    return s.values, c.values
