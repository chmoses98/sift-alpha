def load(kind, cols=None):
    fs = sorted(glob.glob(D + kind + '_*.parquet'))
    return pd.concat([pd.read_parquet(f, columns=cols) for f in fs], ignore_index=True)


def prior_sum(df, col, by=()):
    """Sum of col over rows of df with the same `by` values and date < row's date."""
    by = list(by)
    keys = by + ['date']
    g = df.groupby(keys, sort=True)[col].sum()
    c = (g.groupby(level=by).cumsum() if by else g.cumsum()) - g
    idx = pd.MultiIndex.from_frame(df[keys]) if len(keys) > 1 else pd.Index(df['date'])
    return c.reindex(idx).to_numpy(dtype=float)


def sdiv(a, b):
    a = np.asarray(a, dtype=float); b = np.asarray(b, dtype=float)
    with np.errstate(divide='ignore', invalid='ignore'):
        return np.where(b != 0, a / b, np.nan)


def ewm_pre(df, col, ent, H, m):
    return df.groupby(ent, sort=False)[col].transform(
        lambda s: s.shift(1).ewm(halflife=H, min_periods=m).mean())


def other(df, col):
    s = df.set_index(['game_id', 'team'])[col]
    return s.reindex(pd.MultiIndex.from_frame(df[['game_id', 'opp']])).to_numpy()


def wsum(x, pos, N, H):
    """x ordered by entity (contiguous) then chronologically; pos = index within entity."""
    x = np.asarray(x, dtype=float)
    out = np.zeros(len(x))
    for k in range(1, N + 1):
        idx = np.nonzero(pos >= k)[0]
        if len(idx) == 0:
            break
        out[idx] += x[idx - k] * 0.5 ** (k / H)
    return out

def Fser(a, b, g_, ph, memo):
    if a >= 4: return 1.0
    if b >= 4: return 0.0
    if g_ > 7: return 0.5
    key = (a, b, g_)
    if key in memo: return memo[key]
    p = ph if g_ in (1, 2, 5, 7) else 1 - ph
    v = p * Fser(a + 1, b, g_ + 1, ph, memo) + (1 - p) * Fser(a, b + 1, g_ + 1, ph, memo)
    memo[key] = v
    return v
