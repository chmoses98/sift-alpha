def W(df,cols,key):
    """window sums; df has season, date. key: column name or None. returns DataFrame indexed like df, columns W_<c>"""
    d=df[['season','date']+([key] if key else [])+cols].copy()
    if not key: key='_k'; d[key]=0
    for c in cols: d[c]=d[c].fillna(0).astype('float64')
    d['_i']=np.arange(len(d))
    agg=d.groupby([key,'season','date'],as_index=False)[cols].sum().sort_values([key,'season','date'])
    cs=agg.groupby([key,'season'])[cols].cumsum()-agg[cols]
    cur=agg[[key,'season','date']].copy()
    for c in cols: cur['cur_'+c]=cs[c].values
    tot=d.groupby([key,'season'],as_index=False)[cols].sum()
    tot['season']=tot['season']+1
    tot=tot.rename(columns={c:'prv_'+c for c in cols})
    m=d[['_i',key,'season','date']].merge(cur,on=[key,'season','date'],how='left').merge(tot,on=[key,'season'],how='left').sort_values('_i')
    out=pd.DataFrame(index=df.index)
    for c in cols: out[c]=m['cur_'+c].fillna(0).values+m['prv_'+c].fillna(0).values
    return out
def ratio(a,b):
    a=np.asarray(a,float); b=np.asarray(b,float)
    with np.errstate(all='ignore'):
        return np.where(b>0,a/np.where(b>0,b,1),np.nan)
