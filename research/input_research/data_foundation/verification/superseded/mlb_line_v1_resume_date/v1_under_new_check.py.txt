"""Run the new pit_check_first_pitch_availability check against the VERSION 1 dating (resume date) to show the check detects the defect."""
import sys, json, pickle, types
sys.path.insert(0, "/home/user/sift-alpha")
from alpha_engine.features import mlb_line_ext as ML
from alpha_engine.input_research.data_foundation import replay_generic as G
S = sys.argv[1]; raw = pickle.load(open(f"{S}/mlbfix/raw.pkl", "rb"))
v1 = types.SimpleNamespace(specs=ML.specs, compute=lambda r: ML.compute({"games": r["games"].assign(game_date=r["games"].avail_date).drop(columns="avail_date")}))
r = G.run(v1, raw, "MLB", {"games": ("game_id", ML.CONTENT)}, games_table="games", games_date_col="game_date", quick=True, avail_col="avail_date")
out = {k: v["tests"]["pit_check_first_pitch_availability"] for k, v in r["inputs"].items()}
json.dump({"what": "version 1 (resume-date) builder under the new check; quick mode", "results": out}, open(f"{S}/mlbfix/v1_under_new_check.json", "w"), indent=1)
print({k: (v["ok"], [(c["label"], len(c["bad_columns"])) for c in v["cuts"]]) for k, v in out.items()})
